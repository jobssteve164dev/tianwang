```python
class AIEngineConfig(BaseSettings):
    # AI模型配置 - 重命名以避免与Pydantic受保护命名空间冲突
    ai_model_path: str = "./models"  # 原 model_path
    
    class Config:
```

```python
# server/ai-engine/src/services/kafka_service.py
async def start(self):
    """启动Kafka服务"""
    try:
        # ... 初始化代码 ...
        logger.info("Kafka服务启动成功")
    except Exception as e:
        logger.warning(f"Kafka服务启动失败，将使用离线模式: {e}")
        # 不抛出异常，而是设置为离线模式
        self.is_running = False
        self.consumer = None
        self.producer = None
        logger.info("AI引擎将在离线模式下运行，Kafka功能将不可用")
```
