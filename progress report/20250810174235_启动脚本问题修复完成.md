**端口清理功能**:
```bash
# dev-start.sh
cleanup_port() {
    local port=$1
    local service_name=$2
    
    if lsof -Pi :$port -sTCP:LISTEN -t >/dev/null ; then
        log_warning "端口 $port 被占用，正在清理..."
        
        # 获取占用端口的进程PID
        local pids=$(lsof -ti:$port)
        
        if [ -n "$pids" ]; then
            # 尝试优雅地终止进程
            for pid in $pids; do
                log_info "正在终止进程 $pid (端口 $port)"
                kill $pid 2>/dev/null
            done
            
            # 等待进程终止
            sleep 2
            
            # 检查是否还有进程占用端口
            if lsof -Pi :$port -sTCP:LISTEN -t >/dev/null ; then
                log_warning "优雅终止失败，强制终止进程..."
                for pid in $pids; do
                    kill -9 $pid 2>/dev/null
                done
                sleep 1
            fi
        fi
    fi
}
```
