      setConfig(response.config);
      form.setFieldsValue(response.config);

      message.error('加载配置失败');
    }
  } catch (error) {
    console.error('加载配置失败:', error);
    message.error('加载配置失败');

  }

*   **学到的教训**: 
    - React Hook的依赖数组必须包含所有外部变量，但外部作用域的值（如API对象）不需要包含

    - 组件库的API变更需要及时跟进，避免使用废弃的API
    - 系统性的错误修复方法（从日志分析到根本原因识别）非常有效
