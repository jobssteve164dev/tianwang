```javascript
// 创建不同尺寸的PNG图标
const sizes = [16, 32, 64, 128, 256, 512];
for (const size of sizes) {
    const pngPath = path.join(assetsDir, `icon-${size}.png`);
    await sharp(svgBuffer)
        .resize(size, size)
        .png()
        .toFile(pngPath);
}
```
