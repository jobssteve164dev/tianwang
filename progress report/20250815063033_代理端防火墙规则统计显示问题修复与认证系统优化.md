```javascript
// 检查是否为代理token
if (decoded.type === 'agent') {
  // 代理认证 - 验证代理是否存在且在线
  const agent = await models.Agent.findOne({
    where: { agent_id: decoded.agentId }
  });
  
  if (!agent || agent.status !== 'online') {
    return res.status(401).json({
      error: 'Invalid agent token - agent not found or offline',
      code: 'AGENT_NOT_FOUND'
    });
  }
  
  // 为代理请求创建模拟用户（具有管理员权限）
  req.user = {
    id: 'agent-' + decoded.agentId,
    username: 'agent',
    role: 'admin',
    organization_id: agent.organization_id || '1',
    status: 'active',
    isLocked: () => false,
    isAgent: true,
    agentId: decoded.agentId
  };
  
  logger.debug('代理认证成功:', { agentId: decoded.agentId, hostname: decoded.hostname });
  return next();
}
```

## 5. 自我评估与学习 (Self-Assessment & Learning)

*   **遇到的挑战**: 
    - 最初误解了问题性质，以为是前端显示逻辑问题
    - 需要区分WebSocket认证和HTTP API认证的不同机制
    - 需要理解代理token和用户token的不同用途
*   **学到的教训**: 
    - 在调试网络问题时，日志分析是最直接有效的方法
    - 认证系统的设计需要考虑不同客户端类型（用户vs代理）的不同需求
    - 在修改认证逻辑时需要确保向后兼容性
    - 调试信息的添加对于问题定位至关重要
